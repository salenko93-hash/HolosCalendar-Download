cmake_minimum_required(VERSION 3.22)
project(holos_speech_host LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
find_package(JNI REQUIRED)
add_subdirectory("../../third_party/whisper.cpp" whisper-build)
add_library(holos_whisper SHARED ../../app/src/main/cpp/whisper_jni.cpp)
target_include_directories(holos_whisper PRIVATE ${JNI_INCLUDE_DIRS})
target_link_libraries(holos_whisper PRIVATE whisper)
